cmake_minimum_required(VERSION 3.24)
project(SourceKrillion VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "")
endif()

include(FetchContent)
set(FETCHCONTENT_QUIET ON)
# Every dependency is fetched from source and linked statically: no system libraries needed.
FetchContent_Declare(SDL3 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG release-3.2.26 GIT_SHALLOW TRUE)
FetchContent_Declare(freetype GIT_REPOSITORY https://github.com/freetype/freetype.git GIT_TAG VER-2-13-3 GIT_SHALLOW TRUE)
FetchContent_Declare(lunasvg GIT_REPOSITORY https://github.com/sammycage/lunasvg.git GIT_TAG v3.3.0 GIT_SHALLOW TRUE)
FetchContent_Declare(RmlUi GIT_REPOSITORY https://github.com/mikke89/RmlUi.git GIT_TAG 6.1 GIT_SHALLOW TRUE)
FetchContent_Declare(mbedtls GIT_REPOSITORY https://github.com/Mbed-TLS/mbedtls.git GIT_TAG v3.6.4 GIT_SHALLOW TRUE GIT_SUBMODULES framework)
FetchContent_Declare(IXWebSocket GIT_REPOSITORY https://github.com/machinezone/IXWebSocket.git GIT_TAG v11.4.6 GIT_SHALLOW TRUE)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3 GIT_SHALLOW TRUE)
FetchContent_Declare(canvas_ity GIT_REPOSITORY https://github.com/a-e-k/canvas_ity.git GIT_TAG 8be29cb154d06392385687c721f822d8cce0dc05)
FetchContent_Declare(stb GIT_REPOSITORY https://github.com/nothings/stb.git GIT_TAG 6b5799bae8a1fe3792691ef4b72ae26434c3e799)

# SDL
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(SDL3)

# FreeType without optional system codecs
set(FT_DISABLE_ZLIB ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
set(FT_DISABLE_PNG ON CACHE BOOL "" FORCE)
set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BROTLI ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(freetype)
if(NOT TARGET Freetype::Freetype)
  add_library(Freetype::Freetype ALIAS freetype)
endif()
set(CMAKE_DISABLE_FIND_PACKAGE_Freetype ON)

FetchContent_MakeAvailable(lunasvg)
if(NOT TARGET lunasvg::lunasvg)
  add_library(lunasvg::lunasvg ALIAS lunasvg)
endif()
set(CMAKE_DISABLE_FIND_PACKAGE_lunasvg ON)

set(RMLUI_SVG_PLUGIN ON CACHE BOOL "" FORCE)
set(RMLUI_SAMPLES OFF CACHE BOOL "" FORCE)
set(RMLUI_PRECOMPILED_HEADERS OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(RmlUi)

set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(mbedtls)
set(IXWEBSOCKET_INSTALL OFF CACHE BOOL "" FORCE)
set(USE_TLS ON CACHE BOOL "" FORCE)
set(USE_MBED_TLS ON CACHE BOOL "" FORCE)
set(USE_ZLIB OFF CACHE BOOL "" FORCE)
set(MBEDTLS_INCLUDE_DIRS ${mbedtls_SOURCE_DIR}/include CACHE PATH "" FORCE)
set(MBEDTLS_LIBRARIES mbedtls mbedx509 mbedcrypto CACHE STRING "" FORCE)
set(MBEDTLS_FOUND ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(IXWebSocket)
target_compile_definitions(ixwebsocket PRIVATE IXWEBSOCKET_USE_MBED_TLS_MIN_VERSION_3)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(json)
FetchContent_MakeAvailable(canvas_ity stb)

find_package(OpenGL REQUIRED)
set(RMLUI_BACKENDS ${rmlui_SOURCE_DIR}/Backends)
add_executable(SourceKrillion WIN32 MACOSX_BUNDLE
  src/main.cpp src/app.cpp src/ocean.cpp src/sound.cpp src/net.cpp src/offline.cpp src/content.cpp src/charts.cpp
  ${RMLUI_BACKENDS}/RmlUi_Platform_SDL.cpp ${RMLUI_BACKENDS}/RmlUi_Renderer_GL3.cpp)
target_include_directories(SourceKrillion PRIVATE src ${RMLUI_BACKENDS} ${canvas_ity_SOURCE_DIR}/src ${stb_SOURCE_DIR} ${ixwebsocket_SOURCE_DIR})
target_compile_definitions(SourceKrillion PRIVATE RMLUI_SDL_VERSION_MAJOR=3 SK_VERSION="${PROJECT_VERSION}")
target_link_libraries(SourceKrillion PRIVATE SDL3::SDL3-static RmlUi::RmlUi RmlUi::Debugger ixwebsocket nlohmann_json::nlohmann_json lunasvg::lunasvg OpenGL::GL)
if(WIN32)
  target_sources(SourceKrillion PRIVATE assets/app.rc)
endif()
if(MSVC)
  target_compile_options(SourceKrillion PRIVATE /utf-8 /bigobj)
  target_compile_definitions(SourceKrillion PRIVATE _USE_MATH_DEFINES NOMINMAX)
endif()

# Assets travel next to the executable (or inside the .app bundle on macOS).
if(APPLE)
  set_target_properties(SourceKrillion PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "SourceKrillion" MACOSX_BUNDLE_GUI_IDENTIFIER "io.sourcekrillion.app" MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION} MACOSX_BUNDLE_ICON_FILE "SourceKrillion.icns" MACOSX_BUNDLE_INFO_PLIST ${CMAKE_SOURCE_DIR}/cmake/Info.plist.in)
  set(ASSET_DEST "$<TARGET_BUNDLE_CONTENT_DIR:SourceKrillion>/Resources/assets")
  if(EXISTS ${CMAKE_SOURCE_DIR}/assets/SourceKrillion.icns)
    target_sources(SourceKrillion PRIVATE assets/SourceKrillion.icns)
    set_source_files_properties(assets/SourceKrillion.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  endif()
else()
  set(ASSET_DEST "$<TARGET_FILE_DIR:SourceKrillion>/assets")
endif()
add_custom_target(copy_assets ALL COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/assets ${ASSET_DEST} DEPENDS SourceKrillion)
